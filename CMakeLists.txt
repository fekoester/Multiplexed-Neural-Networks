cmake_minimum_required(VERSION 3.16)
project(MultiplexedNeuralNetworks CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(MNN_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
if(MNN_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra)

add_library(mnn INTERFACE)
target_include_directories(mnn INTERFACE include)
add_executable(mnn_train src/train.cpp)
target_link_libraries(mnn_train PRIVATE mnn)
add_executable(mnn_bench src/bench.cpp)
target_link_libraries(mnn_bench PRIVATE mnn)

enable_testing()
find_path(DOCTEST_INCLUDE doctest/doctest.h)
if(DOCTEST_INCLUDE)
  add_executable(mnn_tests tests/test_mnn.cpp)
  target_include_directories(mnn_tests PRIVATE ${DOCTEST_INCLUDE})
  target_link_libraries(mnn_tests PRIVATE mnn)
  add_test(NAME mnn_tests COMMAND mnn_tests)
endif()
